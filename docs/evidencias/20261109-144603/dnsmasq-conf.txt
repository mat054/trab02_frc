$ cat /usr/local/etc/dnsmasq.conf
# dnsmasq — somente DHCP na LAN. Renderizado por scripts/03-setup-dhcp.sh
# para /usr/local/etc/dnsmasq.conf (os marcadores entre sublinhados duplos vêm de config/lab.env).

# Desliga o DNS do dnsmasq: os clientes recebem DNS_IP via opção 6, não o roteador.
port=0

# Escuta só na LAN. bind-interfaces evita responder DHCP na rede do laboratório.
interface=vtnet1
bind-interfaces
no-dhcp-interface=vtnet0

# Pool dinâmico, máscara /16 e tempo de concessão
dhcp-range=10.20.0.100,10.20.0.200,255.255.0.0,12h

# Gateway e DNS entregues aos clientes
dhcp-option=option:router,10.20.0.1
dhcp-option=option:dns-server,192.168.64.1

# Reserva fixa: MAC do cliente que recebe o DNAT (fora do pool, dentro da LAN)
dhcp-host=52:54:00:4c:41:10,10.20.0.10,cliente-reservado
dhcp-authoritative

# Leases e logs
dhcp-leasefile=/var/db/dnsmasq.leases
log-dhcp
log-facility=/var/log/dnsmasq.log
